import { mutation } from "./_generated/server";
import { v } from "convex/values";
import { distanceKm, estimateEtaSeconds } from "./lib";

/**
 * Moves an ambulance a step closer to its current target (patient location
 * while en_route_to_patient, hospital location while en_route_to_hospital).
 * Called repeatedly on a client-side interval to simulate live GPS movement
 * (Convex doesn't run free-running background loops, so the tick is driven
 * by the UI calling this mutation every N seconds).
 */
export const tickAmbulance = mutation({
  args: { ambulanceId: v.id("ambulances") },
  handler: async (ctx, args) => {
    const ambulance = await ctx.db.get(args.ambulanceId);
    if (!ambulance || !ambulance.currentSessionId) return null;

    const session = await ctx.db.get(ambulance.currentSessionId);
    if (!session) return null;

    let targetLat: number;
    let targetLng: number;

    if (ambulance.status === "en_route_to_patient") {
      targetLat = session.incidentLat;
      targetLng = session.incidentLng;
    } else if (ambulance.status === "en_route_to_hospital") {
      if (!session.assignedHospitalId) return null;
      const hospital = await ctx.db.get(session.assignedHospitalId);
      if (!hospital) return null;
      targetLat = hospital.lat;
      targetLng = hospital.lng;
    } else {
      return null; // idle / with_patient / at_hospital â€” nothing to move
    }

    const remainingKm = distanceKm(ambulance.lat, ambulance.lng, targetLat, targetLng);

    // Move ~12% of the remaining distance per tick â€” fast enough to be a
    // satisfying demo, slow enough to watch it travel across the map.
    const stepFraction = 0.12;
    const arrivalThresholdKm = 0.05;

    if (remainingKm <= arrivalThresholdKm) {
      // Arrived at current target.
      await ctx.db.patch(args.ambulanceId, { lat: targetLat, lng: targetLng });

      if (ambulance.status === "en_route_to_patient") {
        await ctx.db.patch(args.ambulanceId, { status: "with_patient" });
        await ctx.db.patch(ambulance.currentSessionId, { status: "patient_picked_up", etaSeconds: 0 });
        await ctx.db.insert("sessionEvents", {
          sessionId: ambulance.currentSessionId,
          actor: "Ambulance",
          message: `${ambulance.vehicleNumber} has reached the patient.`,
          createdAt: Date.now(),
        });
      } else {
        await ctx.db.patch(args.ambulanceId, { status: "at_hospital" });
        await ctx.db.patch(ambulance.currentSessionId, { status: "arrived_at_hospital", etaSeconds: 0 });
        await ctx.db.insert("sessionEvents", {
          sessionId: ambulance.currentSessionId,
          actor: "Ambulance",
          message: `${ambulance.vehicleNumber} has arrived at the hospital.`,
          createdAt: Date.now(),
        });
      }
      return { arrived: true };
    }

    const newLat = ambulance.lat + (targetLat - ambulance.lat) * stepFraction;
    const newLng = ambulance.lng + (targetLng - ambulance.lng) * stepFraction;
    const newRemainingKm = distanceKm(newLat, newLng, targetLat, targetLng);
    const newEta = estimateEtaSeconds(newRemainingKm);

    await ctx.db.patch(args.ambulanceId, { lat: newLat, lng: newLng });
    await ctx.db.patch(ambulance.currentSessionId, { etaSeconds: newEta });

    return { arrived: false, lat: newLat, lng: newLng, etaSeconds: newEta };
  },
});

import { mutation } from "./_generated/server";
import { v } from "convex/values";
import { distanceKm, estimateEtaSeconds, requiredFacilitiesFor } from "./lib";

// ---------------------------------------------------------------------
// Dispatch the nearest idle ambulance to the incident location.
// Pure proximity â€” every ambulance can reach a scene, so this stays simple.
// ---------------------------------------------------------------------
export const dispatchNearestAmbulance = mutation({
  args: { sessionId: v.id("sessions") },
  handler: async (ctx, args) => {
    const session = await ctx.db.get(args.sessionId);
    if (!session) throw new Error("Session not found");

    const idleAmbulances = await ctx.db
      .query("ambulances")
      .withIndex("by_status", (q) => q.eq("status", "idle"))
      .collect();

    if (idleAmbulances.length === 0) {
      throw new Error("No idle ambulances available right now.");
    }

    let nearest = idleAmbulances[0];
    let nearestDist = distanceKm(
      session.incidentLat,
      session.incidentLng,
      nearest.lat,
      nearest.lng,
    );
    for (const amb of idleAmbulances.slice(1)) {
      const d = distanceKm(session.incidentLat, session.incidentLng, amb.lat, amb.lng);
      if (d < nearestDist) {
        nearest = amb;
        nearestDist = d;
      }
    }

    const etaSeconds = estimateEtaSeconds(nearestDist);

    await ctx.db.patch(nearest._id, {
      status: "en_route_to_patient",
      currentSessionId: args.sessionId,
    });

    await ctx.db.patch(args.sessionId, {
      assignedAmbulanceId: nearest._id,
      etaSeconds,
      status: "ambulance_dispatched",
    });

    await ctx.db.insert("sessionEvents", {
      sessionId: args.sessionId,
      actor: "Dispatch Agent",
      message: `Ambulance ${nearest.vehicleNumber} (${nearest.driverName}) dispatched â€” ${nearestDist.toFixed(1)} km away, ETA ~${Math.round(etaSeconds / 60)} min.`,
      createdAt: Date.now(),
    });

    return { ambulanceId: nearest._id, distanceKm: nearestDist, etaSeconds };
  },
});

// ---------------------------------------------------------------------
// Assign the best hospital based on CONDITION MATCH first, distance second.
// Requires extractedSymptoms to already be on the session (post-intake).
// ---------------------------------------------------------------------
export const assignBestHospital = mutation({
  args: { sessionId: v.id("sessions") },
  handler: async (ctx, args) => {
    const session = await ctx.db.get(args.sessionId);
    if (!session) throw new Error("Session not found");

    const required = requiredFacilitiesFor(session.extractedSymptoms ?? []);
    const allHospitals = await ctx.db.query("hospitals").collect();

    // Score: hospitals that match ALL required facilities AND have a free
    // bed are eligible. Among eligible, pick nearest. If none match fully,
    // fall back to "most facilities matched, then nearest" so we never
    // leave a patient unassigned.
    const withScore = allHospitals
      .filter((h) => h.availableBeds > 0)
      .map((h) => {
        const matched = required.filter((f) => h.facilities.includes(f)).length;
        const dist = distanceKm(session.incidentLat, session.incidentLng, h.lat, h.lng);
        return { hospital: h, matched, fullMatch: matched === required.length, dist };
      })
      .sort((a, b) => {
        if (a.fullMatch !== b.fullMatch) return a.fullMatch ? -1 : 1;
        if (b.matched !== a.matched) return b.matched - a.matched;
        return a.dist - b.dist;
      });

    if (withScore.length === 0) {
      throw new Error("No hospital with available beds found.");
    }

    const best = withScore[0];

    await ctx.db.patch(args.sessionId, {
      assignedHospitalId: best.hospital._id,
    });

    await ctx.db.patch(best.hospital._id, {
      availableBeds: best.hospital.availableBeds - 1,
    });

    await ctx.db.insert("sessionEvents", {
      sessionId: args.sessionId,
      actor: "Dispatch Agent",
      message: `Hospital selected: ${best.hospital.name} â€” matched ${best.matched}/${required.length} required facilities (${required.join(", ")}), ${best.dist.toFixed(1)} km from scene.`,
      createdAt: Date.now(),
    });

    return { hospitalId: best.hospital._id, matched: best.matched, required };
  },
});

/// <reference types="node" />
"use node";

import { action } from "./_generated/server";
import { v } from "convex/values";
import { internal } from "./_generated/api";
import { GoogleGenerativeAI } from "@google/generative-ai";
import { FIRST_AID_STEPS } from "./lib";

const KNOWN_SYMPTOMS = Object.keys(FIRST_AID_STEPS);

function getClient() {
  const apiKey = process.env.GEMINI_API_KEY;
  if (!apiKey) {
    throw new Error(
      "GEMINI_API_KEY is not set. Run `npx convex env set GEMINI_API_KEY <your-key>`.",
    );
  }
  return new GoogleGenerativeAI(apiKey);
}

export const processIntakeTranscript = action({
  args: {
    sessionId: v.id("sessions"),
    transcript: v.string(),
  },
  handler: async (ctx, args) => {
    const genAI = getClient();
    const model = genAI.getGenerativeModel({ model: "gemini-2.5-flash-lite" });

    const prompt = `You are an emergency intake triage assistant listening to a live 911/108-style emergency call transcript.

Known symptom keywords you may use (ONLY use these exact strings, nothing else):
${KNOWN_SYMPTOMS.map((s) => `- ${s}`).join("\n")}

Transcript so far:
"""
${args.transcript}
"""

Return ONLY a JSON array of matching symptom keywords from the list above that are clearly indicated by the transcript. If none are clearly indicated, return an empty array. Do not include any explanation, markdown, or text outside the JSON array.
Example output: ["bleeding", "unconscious"]`;

    let extractedSymptoms: string[] = [];
    try {
      const result = await model.generateContent(prompt);
      const text = result.response.text().trim();
      const cleaned = text.replace(/```json|```/g, "").trim();
      const parsed = JSON.parse(cleaned);
      if (Array.isArray(parsed)) {
        extractedSymptoms = parsed.filter((s) => KNOWN_SYMPTOMS.includes(s));
      }
    } catch (err) {
      console.error("Gemini intake extraction failed, falling back to keyword match:", err);
      const lower = args.transcript.toLowerCase();
      extractedSymptoms = KNOWN_SYMPTOMS.filter((k) => lower.includes(k));
    }

    const firstAidGiven = Array.from(
      new Set(extractedSymptoms.flatMap((s) => FIRST_AID_STEPS[s] ?? [])),
    );

    await ctx.runMutation(internal.sessions.applyIntakeResult, {
      sessionId: args.sessionId,
      transcript: args.transcript,
      extractedSymptoms,
      firstAidGiven,
    });

    return { extractedSymptoms, firstAidGiven };
  },
});

// Shared domain helpers used by multiple Convex functions.

/** Haversine distance in kilometers between two lat/lng points. */
export function distanceKm(
  lat1: number,
  lng1: number,
  lat2: number,
  lng2: number,
): number {
  const R = 6371;
  const dLat = ((lat2 - lat1) * Math.PI) / 180;
  const dLng = ((lng2 - lng1) * Math.PI) / 180;
  const a =
    Math.sin(dLat / 2) ** 2 +
    Math.cos((lat1 * Math.PI) / 180) *
      Math.cos((lat2 * Math.PI) / 180) *
      Math.sin(dLng / 2) ** 2;
  const c = 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
  return R * c;
}

/** Rough ETA in seconds assuming average urban ambulance speed (with siren). */
export function estimateEtaSeconds(km: number): number {
  const avgSpeedKmh = 40;
  return Math.round((km / avgSpeedKmh) * 3600);
}

/**
 * Maps extracted symptom keywords to the hospital facility tags required
 * to treat them. This is intentionally simple/rule-based (not LLM) because
 * facility *requirements* are a known, stable mapping â€” not a judgment call.
 */
const SYMPTOM_TO_FACILITY: Record<string, string[]> = {
  bleeding: ["trauma"],
  "heavy bleeding": ["trauma", "blood_bank"],
  fracture: ["trauma", "orthopedic"],
  unconscious: ["trauma", "neuro"],
  "head injury": ["neuro", "trauma"],
  "not breathing": ["icu", "ventilator"],
  drowning: ["icu", "ventilator"],
  "chest pain": ["cardiac"],
  "heart attack": ["cardiac", "icu"],
  burns: ["burns_unit"],
  seizure: ["neuro"],
  "spinal injury": ["neuro", "trauma"],
  pregnant: ["maternity"],
  "snake bite": ["icu", "antivenom"],
  poisoning: ["icu"],
  stroke: ["neuro", "icu"],
};

export function requiredFacilitiesFor(symptoms: string[]): string[] {
  const set = new Set<string>();
  for (const symptom of symptoms) {
    const key = symptom.toLowerCase().trim();
    const mapped = SYMPTOM_TO_FACILITY[key];
    if (mapped) mapped.forEach((f) => set.add(f));
  }
  // Always sensible defaults if nothing specific matched.
  if (set.size === 0) set.add("trauma");
  return Array.from(set);
}

import { defineSchema, defineTable } from "convex/server";
import { v } from "convex/values";

export default defineSchema({
  // ---------------------------------------------------------------------
  // PATIENTS â€” persistent identity + medical history, independent of any
  // single emergency. Looked up by phone / govt ID / vehicle number / face.
  // ---------------------------------------------------------------------
  patients: defineTable({
    fullName: v.string(),
    phone: v.optional(v.string()),
    govtId: v.optional(v.string()), // Aadhaar-like ID, mocked
    vehicleNumber: v.optional(v.string()),
    age: v.optional(v.number()),
    bloodGroup: v.optional(v.string()),
    knownAllergies: v.optional(v.array(v.string())),
    chronicConditions: v.optional(v.array(v.string())),
    photoUrl: v.optional(v.string()), // for mock face-match
    createdAt: v.number(),
  })
    .index("by_phone", ["phone"])
    .index("by_govtId", ["govtId"])
    .index("by_vehicleNumber", ["vehicleNumber"]),

  // Past hospital visits / records tied to a patient (the "normal app that
  // stores patient records" you described).
  medicalRecords: defineTable({
    patientId: v.id("patients"),
    visitDate: v.number(),
    hospitalId: v.optional(v.id("hospitals")),
    diagnosis: v.string(),
    notes: v.optional(v.string()),
  }).index("by_patient", ["patientId"]),

  // ---------------------------------------------------------------------
  // AMBULANCES â€” static seeded fleet, lat/lng updated on a tick to fake
  // live GPS movement.
  // ---------------------------------------------------------------------
  ambulances: defineTable({
    vehicleNumber: v.string(),
    driverName: v.string(),
    helperName: v.string(),
    lat: v.number(),
    lng: v.number(),
    status: v.union(
      v.literal("idle"),
      v.literal("dispatched"),
      v.literal("en_route_to_patient"),
      v.literal("with_patient"),
      v.literal("en_route_to_hospital"),
      v.literal("at_hospital"),
    ),
    currentSessionId: v.optional(v.id("sessions")),
    equipmentLevel: v.union(
      v.literal("basic"),
      v.literal("advanced"), // ALS - advanced life support
    ),
  }).index("by_status", ["status"]),

  // ---------------------------------------------------------------------
  // HOSPITALS â€” static seeded list with capability flags + live bed/doctor
  // counts so dispatch can pick the *nearest capable* hospital, not just
  // nearest.
  // ---------------------------------------------------------------------
  hospitals: defineTable({
    name: v.string(),
    lat: v.number(),
    lng: v.number(),
    facilities: v.array(v.string()), // e.g. ["trauma", "neuro", "burns", "cardiac"]
    totalBeds: v.number(),
    availableBeds: v.number(),
    erHeadName: v.string(),
  }),

  doctors: defineTable({
    hospitalId: v.id("hospitals"),
    name: v.string(),
    specialty: v.string(),
    available: v.boolean(),
  }).index("by_hospital", ["hospitalId"]),

  // ---------------------------------------------------------------------
  // TRAFFIC SIGNALS â€” seeded points along routes; mock "outpost" that gets
  // notified + flipped green as ambulance approaches.
  // ---------------------------------------------------------------------
  signals: defineTable({
    name: v.string(),
    lat: v.number(),

/**
 * Called on each ambulance movement tick. Any signal within `triggerRadiusKm`
 * of the ambulance's current position gets flipped green + police-notified
 * for this session; signals the ambulance has moved away from revert to red.
 */
export const updateSignalsForAmbulancePosition = mutation({
  args: {
    sessionId: v.id("sessions"),
    ambulanceLat: v.number(),
    ambulanceLng: v.number(),
  },
  handler: async (ctx, args) => {
    const triggerRadiusKm = 1.2;
    const allSignals = await ctx.db.query("signals").collect();

    for (const signal of allSignals) {
      const dist = distanceKm(args.ambulanceLat, args.ambulanceLng, signal.lat, signal.lng);
      const shouldBeGreen = dist <= triggerRadiusKm;

      if (shouldBeGreen && signal.activeSessionId !== args.sessionId) {
        await ctx.db.patch(signal._id, {
          state: "green",
          policeNotified: true,
          activeSessionId: args.sessionId,
        });
        await ctx.db.insert("sessionEvents", {
          sessionId: args.sessionId,
          actor: "Traffic Control Agent",
          message: `Signal "${signal.name}" switched to GREEN, local police outpost notified.`,
          createdAt: Date.now(),
        });
      } else if (!shouldBeGreen && signal.activeSessionId === args.sessionId) {
        await ctx.db.patch(signal._id, {
          state: "red",
          policeNotified: false,
          activeSessionId: undefined,
        });
      }
    }
  },
});

export const resetAllSignals = mutation({

    const model = genAI.getGenerativeModel({ model: "gemini-2.5-flash-lite" });
